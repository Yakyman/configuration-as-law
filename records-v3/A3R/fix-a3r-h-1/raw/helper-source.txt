233:     # as values. Since unit names can't have whitespace in systemd,
234:     # simply use split and strip any leading/trailing quotes. See
235:     # systemd-escape(1) for examples of valid unit names.
236:     open my $fh, '<', "$dpkg_root$service_path" or error("unable to read $dpkg_root$service_path");

244:                 my $wants_dir = "/etc/systemd/$instance/$value";

310: sub enable {
311:     my ($scriptname, $service_path) = @_;
312:     if ($has_systemctl) {
313:         # We use 'systemctl preset' on the initial installation only.
314:         # On upgrade, we manually add the missing symlinks only if the
315:         # service already has some links installed. Using 'systemctl
316:         # preset' allows administrators and downstreams to alter the
317:         # enable policy using systemd-native tools.
318:         my $create_links = 0;
319:         if (debian_installed($scriptname)) {
320:             $create_links = 1 unless no_link_installed($scriptname, $service_path);
321:         } else {
322:             debug "Using systemctl preset to enable $scriptname";
323:             my $systemd_root = '/';
324:             if ($dpkg_root ne '') {
325:                 $systemd_root = $dpkg_root;
326:             }
327:             system("systemctl",
328:                    "--root=$systemd_root",
329:                    $instance eq "user" ? "--global" : "--system",
330:                    "--preset-mode=enable-only",
331:                    "preset", $scriptname) == 0
332:                 or error("systemctl preset failed on $scriptname: $!");
333:         }
334:         make_systemd_links($scriptname, $service_path, create_links => $create_links);
335:     } else {
336:         # We create all the symlinks ourselves
337:         make_systemd_links($scriptname, $service_path);
338:     }

341: sub make_systemd_links {
342:     my ($scriptname, $service_path, %opts) = @_;
343:     $opts{'create_links'} //= 1;
344: 
345:     my $dsh_state = dsh_state_path($scriptname);
346: 
347:     my @links = get_link_closure($scriptname, $service_path);
348:     for my $link (@links) {
349:         my $service_path = $link->{dest};
350:         my $service_link = $link->{src};
351: 
352:         record_in_statefile($dsh_state, $service_link);
353: 
354:         my $statefile = $service_link;
355:         $statefile =~ s,^/etc/systemd/$instance/,$enabled_state_dir/,;
356:         $service_link = "$dpkg_root$service_link";
357:         assertdpkgroot($statefile, "make_systemd_links");
358:         assertdpkgroot($service_link, "make_systemd_links");
359:         assertnotdpkgroot($service_path, "make_systemd_links");
360:         next if -e $statefile;
361: 
362:         if ($opts{'create_links'} && ! -l $service_link) {
363:             make_path(dirname($service_link));
364:             symlink($service_path, $service_link) or
365:                 error("unable to link $service_link to $service_path: $!");
366:             $changed_sth = 1;
367:         }
368: 
369:         # Store the fact that we ran enable for this service_path,
370:         # so that we can skip enable the next time.
371:         # This allows us to call deb-systemd-helper unconditionally
372:         # and still only enable unit files on the initial installation
373:         # of a package.
374:         make_path(dirname($statefile));
375:         open(my $fh, '>>', $statefile) or error("Failed to create/touch $statefile");
376:         close($fh) or error("Failed to create/touch $statefile");
377:     }

441: sub remove_links {
442:     my ($service_path) = @_;
443: 
444:     my $dsh_state = dsh_state_path($service_path);
445:     my @entries = state_file_entries($dsh_state);
446:     debug "Contents: " . Data::Dumper::Dumper(\@entries);
447:     assertdpkgroot($dsh_state, "remove_links");
448:     assertnotdpkgroot($service_path, "remove_links");
449: 
450:     if (is_purge()) {
451:         unlink($dsh_state) if -e $dsh_state;
452:     }
453: 
454:     # Also disable all the units which were enabled when this one was enabled.
455:     for my $link (@entries) {
456:         # Delete the corresponding state file:
457:         # • Always when purging
458:         # • If the user did not disable (= link still exists) the service.
459:         #   If we don’t do this, the link will be deleted a few lines down,
460:         #   but not re-created when re-installing the package.
461:         assertdpkgroot($link, "remove_links");
462:         if (is_purge() || -l $link) {
463:             my $link_state = $link;
464:             $link_state =~ s,^\Q$dpkg_root\E/etc/systemd/$instance/,$enabled_state_dir/,;
465:             unlink($link_state);
466:         }
467: 
468:         next unless -l $link;
469:         unlink($link) or
470:             print STDERR "$0: unable to remove '$link': $!\n";
471: 
472:         $changed_sth = 1;
473:     }
474: 
475:     # Read $service_path, recurse for all Also= units.
476:     # This might not work when $service_path was already deleted,
477:     # i.e. after apt-get remove. In this case we just return
478:     # silently in order to not confuse the user about whether
479:     # disabling actually worked or not — the case is handled by
480:     # dh_installsystemd generating an appropriate disable
481:     # command by parsing the service file at debhelper-time.
482:     open(my $fh, '<', "$dpkg_root$service_path") or return;
483:     while (my $line = <$fh>) {
484:         chomp($line);
485:         my $service_link;
486: 
487:         if ($line =~ /^\s*Also=(.+)$/i) {
488:             remove_links(find_unit($1));
489:         }
490:     }
491:     close($fh);
492: }
